%----------UNIFORM LINE SPACING----------
% Every line - name, section titles, entry rows, bullets, skills - sits exactly
% \RRgap below the one above it, measured from the upper line's descender line to
% the lower line's cap height, whatever the two font sizes are.
%   * each size gets \baselineskip = (cap height + descender) of that size + \RRgap,
%     so wrapped lines inside one paragraph are \RRgap apart;
%   * \RRlineas{<size>} runs right before a new line and shifts \prevdepth by the
%     descender difference between the previous line's size and this one's, so
%     size changes (entry title -> body text, etc.) get the same \RRgap too.
% Computer Modern proportions: cap height 0.683em, descender 0.194em.
\makeatletter
\newdimen\RRprevdesc
\newskip\RRbeforeskip    % space before a section title; refreshed by \RRlineas (titlesec can't take \dimexpr)

\newcommand\RRfixbs{\fontsize{\f@size}{\dimexpr\f@size\p@*877/1000+\RRgap\relax}\selectfont}
\NewCommandCopy\RRorignormalsize\normalsize \DeclareRobustCommand\normalsize{\RRorignormalsize\RRfixbs}
\NewCommandCopy\RRorigsmall\small           \DeclareRobustCommand\small{\RRorigsmall\RRfixbs}
\NewCommandCopy\RRoriglarge\large           \DeclareRobustCommand\large{\RRoriglarge\RRfixbs}
\NewCommandCopy\RRorighuge\huge             \DeclareRobustCommand\huge{\RRorighuge\RRfixbs}

\newcommand\RRlineas[1]{\RRdescof{#1}\par
  \ifdim\prevdepth>-1000pt \prevdepth=\dimexpr\prevdepth-\RRprevdesc+\RRtmpdesc\relax\fi
